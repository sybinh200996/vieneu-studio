Hướng Dẫn Tích Hợp Cho Các Website Khác

Chỉ cần gửi POST request kèm header X-API-Key: <MÃ_KEY> tới endpoint dưới đây:

curl -X POST "https://voiceai.synam.online/api/tts" \
  -H "X-API-Key: vne_live_YOUR_KEY_HERE" \
  -F "text=Xin chào, đoạn âm thanh này được phát từ website của tôi." \
  -F "voice=Hải Đăng" \
  --output speech.wav
const formData = new FormData();
formData.append("text", "Nội dung cần đọc từ website khác");
formData.append("voice", "Thùy Dung"); // Hoặc Quang Sơn, Trúc Ly...

const response = await fetch("https://voiceai.synam.online/api/tts", {
  method: "POST",
  headers: {
    "X-API-Key": "vne_live_YOUR_KEY_HERE"
  },
  body: formData
});

const audioBlob = await response.blob();
const audioUrl = URL.createObjectURL(audioBlob);
new Audio(audioUrl).play();
import requests

url = "https://voiceai.synam.online/api/tts"
headers = {"X-API-Key": "vne_live_YOUR_KEY_HERE"}
data = {
    "text": "Xin chào quý khách, đơn hàng của bạn đã được xác nhận.",
    "voice": "Quang Sơn"
}

response = requests.post(url, headers=headers, data=data)
with open("output.wav", "wb") as f:
    f.write(response.content)
